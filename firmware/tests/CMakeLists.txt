cmake_minimum_required(VERSION 3.20)
project(audiopicture_host_tests LANGUAGES C)
option(AP_ENABLE_SANITIZERS "Enable host Address/UndefinedBehavior sanitizers" OFF)
enable_testing()
add_executable(test_power_governor test_power_governor.c ../components/power_governor/power_governor.c)
target_include_directories(test_power_governor PRIVATE ../components/power_governor/include)
target_compile_features(test_power_governor PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(test_power_governor PRIVATE -Wall -Wextra -Werror)
  if(AP_ENABLE_SANITIZERS)
    target_compile_options(test_power_governor PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_power_governor PRIVATE -fsanitize=address,undefined)
  endif()
endif()
if(UNIX)
  target_link_libraries(test_power_governor PRIVATE m)
endif()
add_test(NAME power_governor_safety_transitions COMMAND test_power_governor)

add_executable(test_environment test_environment.c ../components/environment/ap_environment.c)
target_include_directories(test_environment PRIVATE ../components/environment/include)
target_compile_features(test_environment PRIVATE c_std_11)
target_compile_options(test_environment PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_compile_options(test_power_governor PRIVATE -UNDEBUG)
if(AP_ENABLE_SANITIZERS)
  target_compile_options(test_environment PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(test_environment PRIVATE -fsanitize=address,undefined)
endif()
if(UNIX)
  target_link_libraries(test_environment PRIVATE m)
endif()
add_test(NAME environment_protocol_faults COMMAND test_environment)

add_executable(test_status test_status.c ../components/environment/ap_status.c)
target_include_directories(test_status PRIVATE ../components/environment/include)
target_compile_features(test_status PRIVATE c_std_11)
target_compile_options(test_status PRIVATE -Wall -Wextra -Werror -UNDEBUG)
if(AP_ENABLE_SANITIZERS)
  target_compile_options(test_status PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(test_status PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME status_expander_fail_closed COMMAND test_status)
